window.japp

Native macOS. From JavaScript.

Each call returns a Promise. Swift checks where it came from and whether the manifest allows it.

// a JSON file per app; survives restarts and edits
await japp.storage.set("todos", todos);

// a real Open panel: contents, never the path
const file = await japp.dialog.openFile();

// sent by the app itself: its name, its icon
await japp.notification.show({ title: "Timer done" });

// native HTTP: no CORS, no cookies
const res = await japp.http.fetch(url);

// a real NSMenu with SF Symbols
const choice = await japp.menu.popup([
  { label: "Open", symbol: "arrow.up.forward.square" },
  { label: "Favorite", symbol: "star" }, "-",
  { label: "Delete", symbol: "trash" },
]);
